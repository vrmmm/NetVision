enable
configure terminal

! --- Temel Cihaz Ayarlari ---
hostname FW-01
domain-name netvision.local
enable password Cisco@123
passwd Cisco@123

! --- Banner Mesaji ---
banner motd "===============================================================
  NETVISION - MANAVGAT BELEDIYESI 
  FW-01 CISCO ASA FIREWALL - YETKISIZ ERISIM KESINLIKLE YASAKTIR!
==============================================================="

! --- Dis Ag (Outside / WAN - ISP Baglantisi) Arayuzu ---
interface GigabitEthernet1/1
 nameif outside
 security-level 0
 ip address 203.0.113.2 255.255.255.252
 no shutdown
!

! --- Ic Ag (Inside / LAN - CORE-01 Omurga Baglantisi) Arayuzu ---
interface GigabitEthernet1/2
 nameif inside
 security-level 100
 ip address 10.10.10.3 255.255.255.0
 no shutdown
!

! --- Kullanilmayan Diger Fiziksel Arayuzlerin Kapatilmasi ---
interface GigabitEthernet1/3
 shutdown
!
interface GigabitEthernet1/4
 shutdown
!
interface GigabitEthernet1/5
 shutdown
!
interface GigabitEthernet1/6
 shutdown
!
interface GigabitEthernet1/7
 shutdown
!
interface GigabitEthernet1/8
 shutdown
!

! --- Dinamik NAT / PAT Yapilandirmasi (Tum Ic Ag Icin Dis IP Paylasimi) ---
object network OBJ_CORP_LAN
 subnet 10.10.0.0 255.255.0.0
 nat (inside,outside) dynamic interface
!

! --- Erisim Kontrol Listeleri (Access-Lists - ACL) ---
! Disaridan gelen ICMP cevaplari ve hata mesajlari icin izin
access-list OUTSIDE_IN extended permit icmp any any echo-reply
access-list OUTSIDE_IN extended permit icmp any any time-exceeded
access-list OUTSIDE_IN extended permit icmp any any unreachable
access-group OUTSIDE_IN in interface outside

! --- Moduler Politika Cercevesi (MPF) & ICMP Denetimi (Inspection) ---
! ASA'nin ping/traceroute istek ve yanitlarini durum bilgisiyle (stateful) takip etmesi
policy-map global_policy
 class inspection_default
  inspect icmp
  inspect dns
  inspect http
!
service-policy global_policy global

! --- Statik Yonlendirme (Routing) Tablosu ---
! 1. Default Route: Dis ag / Internet trafigini ISP Router'a yonlendir
route outside 0.0.0.0 0.0.0.0 203.0.113.1 1

! 2. Return Route: Ic ag alt aglarina (10.10.0.0/16) donus trafigini CORE-01'e yonlendir
route inside 10.10.0.0 255.255.0.0 10.10.10.2 1

! --- Yonetim Erisim Guvenligi (SSH) ---
crypto key generate rsa modulus 2048
ssh 10.10.10.0 255.255.255.0 inside
ssh timeout 10

write memory
